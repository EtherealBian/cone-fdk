cmake_minimum_required(VERSION 3.24)
project(cone_fdk LANGUAGES CXX CUDA)
find_package(CUDAToolkit REQUIRED)
add_library(fdk STATIC src/fdk.cu src/filter.cu src/backproject.cu)
target_include_directories(fdk PUBLIC src)
target_link_libraries(fdk PUBLIC CUDA::cudart CUDA::cufft)
target_compile_features(fdk PUBLIC cxx_std_17 cuda_std_17)
add_executable(cone-fdk examples/reconstruct.cpp)
target_link_libraries(cone-fdk PRIVATE fdk)
foreach(stage filter backproject)
  add_executable(bench_${stage} bench/bench_${stage}.cu)
  target_link_libraries(bench_${stage} PRIVATE fdk)
endforeach()
enable_testing()
add_test(NAME dense-demo COMMAND cone-fdk)
find_package(Python3 COMPONENTS Interpreter NumPy QUIET)
if(Python3_Interpreter_FOUND AND Python3_NumPy_FOUND)
  add_test(NAME dense-oracle COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/dense.py $<TARGET_FILE:cone-fdk>)
else()
  message(STATUS "NumPy not found: dense-oracle test unavailable")
endif()
